## Definindo uma funcao para calcular \nu^q(G) quando G \eh um grupo nilpotente 
## dado por uma apresentacao policiclica (por exemplo tomando G um grupo 
## da colecao Small Groups) e q \eh um inteiro nao negativo. 
## 
##

qts:=
	function(G,q) 

local

	iso,       # Isomorfismo que torna o PcGroup G num FpGroup
	fg,        # Imagem de G pelo isomorfismo "iso"
	eg,        # Elementos do Grupo G
	egs,       # Imagem de eg em fg
	f,         # free group (\hat(K))
	fp,        # Produto livre G*G^{\varphi} 
	pgs,       # imagem dos geradores "gen" de G em fg por "iso" 
	I1,	
	i1, i2, 
	I3, i3,    # Imersoes de lg no produto livre  
	lg,        # imagem dos respectivos geradores de pgs na imersao de G em fp
	rg,        # imagem dos respectivos geradores de pgs na imersao de G^{\varphi} em fp
	leg,       # imagem de eg na imersao de G em fp
  reg,       # imagem de eg na imersao de G^{\varphi} em fp
	kg,        # imagem dos respectivos elementos de lg em fp 
	pares,     # pares de geradores [g_i, (g_i)^{\varphi}], com g_i \in lg 
	nurel,     # funcao para produzir as relacoes definidoras para \nu(G) 
	nug,       # o grupo \nu(G)
	relnu,     # relacoes para \nu(G)
	epi,       # epimorfismo do produto livre G*G^{\vfi} em \nu(G) 
	tens,      # quadrado tensorial nao-abeliano de G ( = G \otimes^0 G)
	pl,        # produto livre \nu(G)*f
	r1, r2,    #
	r3, r4,    #
	r5, r6,    # relacoes de q-tensor(G)
	relnuq,    # relacoes exclusivas do quadrado tensorial = Concatenation(r1,...,r6) 
	Relations, # relacoes definidoras de \nu^q(G) = Concatenation(relnu,relnuq) 	
	nuq,       # o grupo \nu^q(G)
	low,       # LowerCentralSeries(nuq);
	llow,      # Length(low) + 1;
	slg,       # imagem de lg em \nu(G)
	srg,       # imagem de rg em \nu(G)
 	sleg,      # imagem de leg em \nu(G)
  	sreg,      # imagem de reg em \nu(G)
	SLG,       # imagem de G em nug 
	SRG,       # imagem de G^{\varphi} em nuq (=\nu(G))
	theta,     # GroupHomomorphismByImages(fp,nuq,GeneratorsOfGroup(fp),GeneratorsOfGroup(nuq))
	epinq,     #### EpimorphismNilpotentQuotient(nuq, NilpotencyClassOfGroup(nuq)+1)
	pcpnuq,    #### Image(epinq) 
	compnq,    #### theta*epinq
	classnuq,  # NilpotencyClassOfGroup(nuq)
	pcpkappa,  ### Image(compnq, Image(Embedding(pl,1)))
	pcpLG,     #### Subgroup(Image(compnq), List(slg, x-> Image(compnq), 			Image(Embedding(pl,1),x)))) 
	pcpSLEG,   #List(sleg, x-> Image(compnq, Image(Embedding(pl,1),x)))
	genpcpLG,  #GeneratorsOfGroup(pcpLG)
	pcpRG,     #Subgroup(Image(compnq), List(srg, x-> Image(theta, 	Image(Embedding(pl,1),x)))) 
	pcpSREG,   #List(sreg, x-> Image(compnq, Image(Embedding(pl,1),x))) 
	genpcpRG,  #GeneratorsOfGroup(pcpRG)
	pcpK,      #Image( compnq,  Image(Embedding(pl,2))   ) 
	genpcpK,   #GeneratorsOfGroup(pcpK)
	pcpT,      #Image(compnq, Image(Embedding(pl,1), tens))
	pcpLGRG,   #CommutatorSubgroup(pcpLG,pcpRG)
	genpcpLGRG, #GeneratorsOfGroup(pcpLGRG)
	pcpqtens,  #Subgroup(pcpnuq, Concatenation(genpcpLGRG , pcpgenK)  )
	pcpDeltaq, #Subgroup(pcpqtens, List([1..Length(pcpSLEG)], i -> 	Comm(pcpSLEG[i], pcpSREG[i])))
	pcpqext,   #FactorGroup(pcpqtens, pcpDeltaq) = ##Quadrado Exterior de G
	pcpnuqkappa,  #FactorGroup(pcpnuq, pcpkappa)
	LG,        # Image(  theta,  Image(Embedding(fp,1))   ) 
	genLG,     # GeneratorsOfGroup(LG);
	RG,        # Image(  theta,  Image(Embedding(fp,2))   ) 
	genRG,     # GeneratorsOfGroup(RG);
	K,         # Image(  theta,  Image(Embedding(fp,3))   ) 
	genK,      # GeneratorsOfGroup(K)
	LGRG,      # CommutatorSubgroup(LG,RG) 
	genLGRG,   # GeneratorsOfGroup(LGRG)
	qt,        # o q-tensor square de G
	inter,     # intersecao de K com LGRG
	resfinal,  # uma lista de resultado
	kappa,     # Image(theta, Image(Embedding(pl,1))) (=imagem de \nu(G) sob theta)
	T,         # Image(theta, Image(Embedding(pl,1), tens))) 
	           # (=imagem do quadrado tensorial de G sob theta)
	qtens,     # Group(  Concatenation(genLGRG , genK)  )
	qtensT,    # qtens/T
	nuqkappa,  # nuq/kappa
	SLEG,      # Image(theta), Elementos de LG
  	SREG,      # Image(theta), Elementos de RG
 	Deltaq,    # \Delta^q(G)
 	 qext;     # q-exterior square of G


	
###################### 

iso:=IsomorphismFpGroupByPcgs(Pcgs(G), "G");
fg:=Image(iso);
eg:=Elements(G); 

pgs:=List(Pcgs(G), x -> Image(iso,x));  

egs:=List(eg, x -> Image(iso,x));

fp:=FreeProduct([fg,fg]);

lg:=List(pgs, x-> Image(Embedding(fp,1),x)); 
### o mesmo que List(pgs, x -> i1(x));  

leg:=List(egs, x -> Image(Embedding(fp,1),x));

rg:=List(pgs, x-> Image(Embedding(fp,2),x)); 
### o mesmo que List(pgs, x -> i2(x));  

reg:=List(egs, x -> Image(Embedding(fp,2),x));

pares:=List([1..Length(lg)], i -> [lg[i],rg[i]]);

nurel:=function(x,a,b,i)
 return Comm(a,b)^x[i]/Comm(a^x[1],b^x[2]); 
end;

relnu:=ListX(pares,lg,rg,[1,2],nurel); 
##isto calcula todas as relacoes s1 e s2 de \nu(G)

nug:=FactorGroupFpGroupByRels(fp, relnu);

epi:= GroupHomomorphismByImages(fp,nug,GeneratorsOfGroup(fp),GeneratorsOfGroup(nug));

slg:=List(lg, x-> Image(epi, x)); 
sleg:=List(leg, x-> Image(epi, x)); 

srg:=List(rg, x-> Image(epi, x));
sreg:=List(reg, x-> Image(epi, x));


SLG:=Subgroup(Image(epi),  slg);

SRG:=Subgroup(Image(epi),  srg);

tens:=CommutatorSubgroup(SLG,SRG); 

f:=FreeGroup(Size(egs));

pl:=FreeProduct(nug, f);

I1:=Embedding(pl,1);

i1:=function(x); 
return Image(I1, Image(epi, Image(Embedding(fp,1),x))); 
end;

i2:=function(x); 
return Image(I1, Image(epi, Image(Embedding(fp,2),x))); 
end;


I3:=Embedding(pl,2); 
i3:=function(x); return Image(I3, f.(Position(egs,x))); end;

## kg:=List(egs, x -> UnderlyingElement(Image(Embedding(fp,3),x)));
### o mesmo que List(egs, x -> i3(x));  


############
##
## Agora as relacoes especificas para o q-tensor product
##
#############    r1: com apenas os geradores de fg,isto \eh , usando os elementos de pgs
##
r1:=function(G,q) 

	local r, x, y;

r:=[];

for x in egs do
	for y in pgs do

	Add( r, 
               i3(x)^i1(y)/i3(x^y) 
            );

	od;
od;

return r; 
end; 

############## r2: com apenas os geradores de fg,isto \eh , 
##usando os elementos de pgs
##

r2:=function(G,q) 

	local r, x, y; 

r:=[];

for x in egs do 
	for y in pgs do

	Add( r, 
              i3(x)^i2(y)/i3(x^y) 
            );
	od;
od;

return r;  
end;


############# r3: com apenas os geradores de fg para i1 e i2, isto \eh , usando os 
############# elementos de pgs dentro do comutador
##
r3:=function(G,q) 
	local r, x, y, z;

r:=[];

for x in pgs do
	for y in pgs do
		for z in egs do

		Add(r,
                     Comm(i1(x),i2(y))^i3(z)/Comm(i1(x^(z^q)),i2(y^(z^q)))
                    );
		od;
	od;
od;

return r;   
end;

############# agora a r4: 
####

r4:=function(G,q)
	
	local r, x, y, l;

r:=[];

for x in egs do
    for y in egs do

	l:= List( [1..(q-1)], i -> Comm( i1(x), i2( (y^-i)^(x^(q-1-i) ) ) ) );
	
        if l<>[] then
 
 	  Add(r, i3(x)*Product(l)*i3(y)/i3(x*y));
 	      else
		Add(r,i3(x)*i3(y)/i3(x*y));
        fi;
       		
	od;
od;

return r; 
end;


######################## Relações do tipo R5 
###

r5:=function(G,q) 

	local r, x, y;

r:=[];

for x in egs do
	for y in egs do

	Add( r,
              Comm(i3(x),i3(y)) / Comm(i1(x^q),i2(y^q))  
            );

	od;
od;

return r; 
end; 

########################## Relações do tipo R6
##
r6:=function(G,q)

	local r, x, y;

r:=[];

for x in egs do
	for y in egs do

	Add( r,
         i3( Comm(x,y) ) / Comm(i1(x),i2(y))^q
            );

	od;
od;

return r;  
end;

############

relnuq:=Concatenation(r1(G,q), r2(G,q), r3(G,q), r4(G,q), r5(G,q), r6(G,q) );

nuq:=FactorGroupFpGroupByRels(pl,relnuq);

low:=LowerCentralSeries(nuq);

llow:=Length(low)+1;

## classnuq:=NilpotencyClassOfGroup(nuq);

epinq:=EpimorphismNilpotentQuotient(nuq, llow);

pcpnuq:=Image(epinq);
 
################

theta:= GroupHomomorphismByImagesNC(pl,nuq,GeneratorsOfGroup(pl),GeneratorsOfGroup(nuq));

compnq:=theta*epinq;

## kappa:=Image(theta, Image(Embedding(pl,1))); ##esta eh a imagem de \nu(G) em \nu^q(G)

pcpkappa:=Image(compnq, Image(Embedding(pl,1))); 

## esta eh a imagem de \nu(G) em pcp\nu^q(G)


## LG:=Subgroup(Image(theta), List(slg, x-> Image(theta, Image(Embedding(pl,1),x)))); 

pcpLG:=Subgroup(Image(compnq), List(slg, x-> Image(compnq, Image(Embedding(pl,1),x)))); 

## SLEG:=List(sleg, x-> Image(theta, Image(Embedding(pl,1),x))); 

pcpSLEG:=List(sleg, x-> Image(compnq, Image(Embedding(pl,1),x))); 

## Estes são os elementos da imersão de G em \nu^q(G) 

## genLG:=GeneratorsOfGroup(LG);

genpcpLG:=GeneratorsOfGroup(pcpLG);

## RG:=Subgroup(Image(theta), List(srg, x-> Image(theta, Image(Embedding(pl,1),x)))); 

pcpRG:=Subgroup(Image(compnq), List(srg, x-> Image(compnq, Image(Embedding(pl,1),x)))); 

## SREG:=List(sreg, x-> Image(theta, Image(Embedding(pl,1),x))); 

pcpSREG:=List(sreg, x-> Image(compnq, Image(Embedding(pl,1),x))); 

##Estes são os elementos da imersão de G^{\varphi} em \nu^q(G)

## genRG:= GeneratorsOfGroup(RG);

genpcpRG:=GeneratorsOfGroup(pcpRG);


K:=Image(  theta,  Image(Embedding(pl,2))   ); 

pcpK:=Image( compnq,  Image(Embedding(pl,2))   ); 


genK:=GeneratorsOfGroup(K);

genpcpK:=GeneratorsOfGroup(pcpK);


T:=Image(theta, Image(Embedding(pl,1), tens));

pcpT:=Image(compnq, Image(Embedding(pl,1), tens));


pcpLGRG:=CommutatorSubgroup(pcpLG,pcpRG); 

## LGRG:=CommutatorSubgroup(LG,RG); 

## genLGRG:=GeneratorsOfGroup(LGRG); 

genpcpLGRG:=GeneratorsOfGroup(pcpLGRG); 

## qtens:= Subgroup(nuq, Concatenation(genLGRG , genK)  );

pcpqtens:= Subgroup(pcpnuq, Concatenation(genpcpLGRG , genpcpK)  );


## Deltaq:=Subgroup(qtens, List([1..Length(SLEG)], i -> Comm(SLEG[i], SREG[i])));

pcpDeltaq:=Subgroup(pcpqtens, List([1..Length(pcpSLEG)], i -> Comm(pcpSLEG[i], pcpSREG[i])));

## qtensT:=FactorGroup(qtens, T);

pcpqext:=FactorGroup(pcpqtens, pcpDeltaq); 

##Quadrado Exterior de G

###nuqkappa:=FactorGroup(pcpnuq, pcpkappa);

pcpnuqkappa:=FactorGroup(pcpnuq, pcpkappa);

inter:=Intersection(pcpLGRG,pcpK); 

resfinal:=[["nu(G)=", nug, Size(nug)], [SLG, SRG, tens, Size(tens)],   
["nuq(g,q)=", pcpnuq, Collected(Factors(Size(pcpnuq))), "class=", llow-2], ["qtens=", pcpqtens, AbelianInvariants(pcpqtens)], ["qext=", pcpqext, AbelianInvariants(pcpqext)], ["Deltaq=", pcpDeltaq, AbelianInvariants(pcpDeltaq)], ["K=", pcpK, Collected(Factors(Size(pcpK)))], 
["T=", pcpT, Collected(Factors(Size(pcpT)))], ["intersection =", inter, Size(inter)], ["nuq/nug=", pcpnuqkappa, Size(pcpnuqkappa)]];

return resfinal; 

end;
